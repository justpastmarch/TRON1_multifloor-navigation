 w0=min(wheel,key=lambda w:abs(w['stamp']-p0['stamp']));w1=min(wheel,key=lambda w:abs(w['stamp']-p1['stamp']));wy=lambda w:Rotation.from_quat(w['quat']).as_euler('xyz')[2]

 out.append(d)
 print(json.dumps({k:v for k,v in d.items() if k!='grid'},indent=2));print('grid passes',[(x['v'],round(x['w'],3)) for x in grid if x['accepted']])
